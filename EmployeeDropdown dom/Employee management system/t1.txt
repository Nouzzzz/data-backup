// Renamed 'name' to 'empName' to avoid conflicts with window.name
let empName = document.getElementById("name") 
let uid = document.getElementById("uid")
let position = document.getElementById("position")
let salary = document.getElementById("salary")
let btn = document.getElementById("btn")
let updateBtn = document.getElementById("updateBtn")
let deleteBtn = document.getElementById("deleteBtn")

// FIX: Ensure you select your table element from the HTML
let table = document.getElementById("table") 

// Uncommented this so your data actually persists across page refreshes
let employees = JSON.parse(localStorage.getItem('employees')) || []
let empindex = -1

function updatestorage () {
    localStorage.setItem('employees', JSON.stringify(employees))
}

function addEmployee() {
    let employee = {
        name: empName.value,
        uid: uid.value,
        position: position.value,
        salary: salary.value
    }

    employees.push(employee)
    updatestorage()
}

function clearfields() {
    empName.value = ""
    uid.value = ""
    position.value = ""
    salary.value = ""
}

function display() {
    // Reset table headers
    table.innerHTML = `
        <tr>
            <th>Name</th>
            <th>Unique id</th>
            <th>Position</th>
            <th>Salary</th>
            <th>Actions</th>
        </tr>
    `

    // Populate rows
    for (let i = 0; i < employees.length; i++) {
        table.innerHTML += `
           <tr>
                <td><p>${employees[i].name}</p></td>
                <td><p>${employees[i].uid}</p></td>
                <td><p>${employees[i].position}</p></td>
                <td><p>${employees[i].salary}</p></td>
                <td> 
                     <!-- FIX: Removed static IDs and linked delete button to delete function -->
                     <button class="edit-btn" onclick="setEdit(${i})">Edit</button>
                     <button class="delete-btn" onclick="deleteEmployee(${i})">Delete</button> 
                </td>
           </tr>
        `
    }
}

function setEdit(index) {
    empindex = index

    empName.value = employees[index].name
    uid.value = employees[index].uid
    position.value = employees[index].position
    salary.value = employees[index].salary

    btn.style.display = "none"
    updateBtn.style.display = "inline-block"
}

function updateEmployee() {
    if (empindex === -1) return;

    employees[empindex] = {
        name: empName.value,
        uid: uid.value,
        position: position.value,
        salary: salary.value
    }

    // FIX: Save updated array to local storage
    updatestorage() 
    display()
    clearfields()

    btn.style.display = "inline-block"
    updateBtn.style.display = "none"
    empindex = -1
}

// FIX: Rewrote completely to properly remove the employee item
function deleteEmployee(index) {
    // Remove 1 item from the specified index
    employees.splice(index, 1) 
    
    updatestorage()
    display()
    clearfields()

    // Reset button visibility states just in case a user was mid-edit
    btn.style.display = "inline-block"
    updateBtn.style.display = "none"
    empindex = -1
}

// Event Listeners
btn.addEventListener('click', () => {
    if(!empName.value || !uid.value) return alert("Please fill out the form");
    addEmployee()
    clearfields()
    display()
})

updateBtn.addEventListener('click', () => {
    updateEmployee()
})

display()
