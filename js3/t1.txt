let students = []

function addStudent () {
    let student = {
        name:name.value,
        age:age.value,
        place:place.value
    }

    students.push(student)

}


function clearfields(){
    name.value = ""
    age.value = ""
    place.value = ""
}

                <p>${students[i].place}</p>

            </div>
        `
    }
}

let name = document.getElementById("name")
let age = document.getElementById("age")
let place = document.getElementById("place")
let btn = document.getElementById("btn")
let itemcontainer = document.getElementById("itemcontainer")

let students = []

function addStudent() {
    let student = {
        name: name.value,
        age: age.value,
        place: place.value
    }

    students.push(student)

}

function clearfields() {
    name.value = ""
    age.value = ""
    place.value = ""
}

function display() {
    itemcontainer.innerHTML = ""
    for (let i = 0; i < students.length; i++) {
        itemcontainer.innerHTML += `
            <div>
           
           

<table>
  <tr>
    <th>Name</th><br>
    <th>Age</th><br>
    <th>Place</th>
  </tr>
  <tr>
    <td> <p>${students[i].name}</p></td>
    <td><p>${students[i].age}</p></td>
    <td><p>${students[i].place}</p></td>
  </tr>
</table>

            </div>
        `
    }
}


btn.addEventListener('click', () => {
    addStudent()
    clearfields()
    display()
})
